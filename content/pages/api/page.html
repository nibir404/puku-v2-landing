Example cURL API Request

# Create an autonomous agent task
curl -X POST https://api.puku.ai/v1/tasks \
-H "Authorization: Bearer $PUKU_API_KEY" \
-H "Content-Type: application/json" \
-d '{
"prompt": "Fix bug in user authentication token refresh flow",
"repo": "organization/backend-service",
"auto_pr": true
}'
